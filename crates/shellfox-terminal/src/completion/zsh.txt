# Shellfox completion. Load after compinit.
_shellfox_complete() {
  local output name pattern='^[A-Za-z][A-Za-z0-9_-]{0,63}$'
  local -a names subcommands
  if (( CURRENT == 2 )); then
    subcommands=('host:Run or manage the host' 'session:Manage sessions' 'run:Run an automation process with pipes' 'runs:Inspect or stop automation runs' 'terminals:Manage and attach terminals' 'network:Manage Tailscale networking' 'machines:Manage remote machines' 'approvals:Manage access approvals' 'settings:Read or update settings' 'profiles:List terminal profiles' 'path:Use host path helpers' 'commands:Run or manage script commands' 'ssh:Connect to a saved SSH profile' 'api:Call a validated manager method' 'remote:Call remote management' 'update:Update Shellfox' 'version:Print the version' 'help:Show help' 'completion:Print shell completion' '--host:Select a local or remote host' '--json:Print JSON' '--yes:Confirm destructive operations')
    _describe 'Shellfox command' subcommands
    return 0
  fi
  if [[ $words[2] != cmd && $words[2] != commands ]]; then
    (( CURRENT == 3 )) || return 0
    case "$words[2]" in
      host) subcommands=(run status stop service keep-awake) ;;
      network) subcommands=(enable disable login status) ;;
      machines) subcommands=(list connect disconnect forget) ;;
      approvals) subcommands=(list approve revoke reject) ;;
      session) subcommands=(list create inspect rename pin env settle unsettle delete stats processes focus refresh clear-error) ;;
      ss) subcommands=(list create inspect rename pin env settle unsettle delete stats processes focus refresh clear-error) ;;
      sessions) subcommands=(list create inspect rename pin env settle unsettle delete stats processes focus refresh clear-error) ;;
      terminals) subcommands=(list create rename reorder close attach write) ;;
      tm) subcommands=(list create rename reorder close attach write) ;;
      run) subcommands=(--cwd --title --request-id --detach --) ;;
      runs) subcommands=(list show stop) ;;
      update) subcommands=(--check) ;;
      settings) subcommands=(read update) ;;
      profiles) subcommands=(list) ;;
      path) subcommands=(home resolve complete create) ;;
      remote) subcommands=(api) ;;
      *) return 0 ;;
    esac
    compadd -- "${subcommands[@]}"
    return 0
  fi
  if (( CURRENT == 3 )); then
    local -a operations
    operations=('--run:Run a registered command' '--register:Register a name and script path' '--ls:List registered commands' '--list:List registered commands' '--remove:Remove a registered command' '--rm:Remove a registered command' '--unregister:Remove a registered command' '--rename:Rename a registered command' '--help:Show custom command help' '-h:Show custom command help')
    _describe 'Custom command operation' operations
  elif (( CURRENT == 4 )); then
    case "$words[3]" in --run|--remove|--rm|--unregister|--rename) ;; *) return 0 ;; esac
  else
    return 0
  fi
  [[ $PREFIX != -* ]] || return 0
  output=$("$words[1]" cmd --complete "$PREFIX" 2>/dev/null) || return 0
  for name in "${(@f)output}"; do
    name=${name%$'\r'}
    # Names are ASCII identifiers. Do not pass arbitrary endpoint text to compadd.
    if [[ $name =~ $pattern ]]; then names+=("$name"); fi
  done
  (( ${#names} )) && compadd -Q -M 'm:{a-zA-Z}={A-Za-z}' -X 'Registered command' -- "${names[@]}"
  return 0
}
(( $+functions[compdef] )) && compdef _shellfox_complete shellfox
